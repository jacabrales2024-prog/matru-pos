📋 Código para Google Apps Script (5 TB Drive)

Pega este código en script.google.com, implementa como Aplicación Web (Acceso: Cualquier usuario) y copia la URL resultante en el conector de Drive:

function doPost(e) {
  var folderName = 'MATRU_POS_Respaldos';
  var folders = DriveApp.getFoldersByName(folderName);
  var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
  var data = JSON.parse(e.postData.contents);
  if (data.type === 'pdf' && data.base64) {
    var blob = Utilities.newBlob(Utilities.base64Decode(data.base64), 'application/pdf', data.fileName);
    var file = folder.createFile(blob);
    return ContentService.createTextOutput(JSON.stringify({ ok: true, url: file.getUrl() })).setMimeType(ContentService.MimeType.JSON);
  }
  var fileName = 'matru_db_backup.json';
  var files = folder.getFilesByName(fileName);
  if (files.hasNext()) {
    files.next().setContent(JSON.stringify(data.payload, null, 2));
  } else {
    folder.createFile(fileName, JSON.stringify(data.payload, null, 2), MimeType.PLAIN_TEXT);
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
}
